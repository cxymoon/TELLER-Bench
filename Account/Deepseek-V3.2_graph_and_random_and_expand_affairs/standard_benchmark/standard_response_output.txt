def parse_application_data(raw_data_list):
    """Parse raw application data strings into dictionaries"""
    applications = []
    
    for raw_entry in raw_data_list:
        app_dict = {}
        
        # Split by field separators while respecting nested structures
        current_field = ""
        current_value = ""
        in_nested = 0
        parsing_key = True
        
        i = 0
        while i < len(raw_entry):
            char = raw_entry[i]
            
            if char in ['{', '[']:
                in_nested += 1
                current_value += char
            elif char in ['}', ']']:
                in_nested -= 1
                current_value += char
            elif char == ':' and in_nested == 0 and parsing_key:
                current_field = current_value.strip()
                current_value = ""
                parsing_key = False
            elif char == ',' and in_nested == 0 and i + 1 < len(raw_entry) and raw_entry[i + 1] == ' ':
                # Field boundary
                if current_field:
                    value = current_value.strip()
                    # Type conversion
                    if value == '':
                        app_dict[current_field] = ''
                    elif value == 'True':
                        app_dict[current_field] = True
                    elif value == 'False':
                        app_dict[current_field] = False
                    elif value.startswith('[') or value.startswith('{'):
                        try:
                            app_dict[current_field] = json.loads(value.replace("'", '"'))
                        except:
                            app_dict[current_field] = value
                    elif '.' in value and value.replace('.', '').replace('-', '').isdigit():
                        app_dict[current_field] = float(value)
                    elif value.replace('-', '').isdigit() and '-' in value and len(value.split('-')) <= 3:
                        app_dict[current_field] = value  # Keep as string (e.g., SSN, EIN)
                    elif value.isdigit():
                        app_dict[current_field] = int(value)
                    else:
                        app_dict[current_field] = value
                
                current_field = ""
                current_value = ""
                parsing_key = True
                i += 1  # Skip space
            else:
                current_value += char
            
            i += 1
        
        # Handle last field
        if current_field:
            value = current_value.strip()
            if value == '':
                app_dict[current_field] = ''
            elif value == 'True':
                app_dict[current_field] = True

            elif value.isdigit():
                app_dict[current_field] = int(value)

                app_dict[current_field] = value
        
        applications.append(app_dict)
    
    return applications
