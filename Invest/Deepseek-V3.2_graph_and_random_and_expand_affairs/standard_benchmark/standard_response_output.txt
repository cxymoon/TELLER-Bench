def parse_entry(entry_list: List[str]) -> Dict[str, Any]:
    """Parse a data entry from list format to dictionary"""
    entry_dict = {}
    for item in entry_list:
        if ':' in item:
            key, value = item.split(':', 1)
            entry_dict[key] = value
    return entry_dict

def safe_float(value: str, default: float = 0.0) -> float:
    """Safely convert string to float"""
    try:
        if value == '' or value is None:
            return default
        return float(value)
    except:
        return default

def safe_int(value: str, default: int = 0) -> int:
    """Safely convert string to int"""
    try:
        if value == '' or value is None:
            return default
        return int(value)
    except:
        return default

def safe_bool(value: str, default: bool = False) -> bool:
    """Safely convert string to bool"""
    if value == '' or value is None:
        return default
    return value.lower() in ['true', '1', 'yes']

def safe_json(value: str, default: Any = None) -> Any:
    """Safely parse JSON string"""
    try:
        if value == '' or value is None:
            return default if default is not None else {}
        return json.loads(value)
    except:
        return default if default is not None else {}

    }
    
    try:

        transaction_id = entry.get('transaction_id', '')

                client_liquidity_needs=client_liquidity_needs,

            client_authorization_method = entry.get('client_authorization_method', '')
            
            activation_result = AuthorizeAndActivateRiskProfile(

                calculated_risk_profile=calculated_risk_profile,
                client_authorization_method=client_authorization_method,
                transaction_timestamp=transaction_timestamp
            )
            
            risk_profile_expiry_date = activation_result['risk_profile_expiry_date']
            investable_status = activation_result['investable_status']
            
            # Update result for Risk_Profiling transactions

                'risk_profile_expiry_date': risk_profile_expiry_date,
                'investable_status': investable_status,
                'suitability_check_passed': '',
                'suitability_mismatch_flag': '',
                'pre_trade_compliance_passed': '',
                'funds_availability_verified': '',
                'estimated_units': '',
                'front_end_load_amount': '',
                'net_investment_amount': '',
                'annual_management_fee_amount': '',
                'trade_reference_number': '',
                'contract_note_number': '',
                'transaction_status': 'Completed',
                'execution_timestamp': '',
                'settlement_date': '',
                'straight_through_processing_flag': 'True',
                'compliance_defect_flag': 'False',
                'compliance_review_notes': 'Risk profiling completed successfully',

            # Step 5.3.1: Execute AML Sanctions Screening

            )
            
            pre_trade_compliance_passed = compliance_aggregate['pre_trade_compliance_passed']
            
            # Step 5.3.5: Calculate Transaction Financials
            front_end_load_percentage = safe_float(entry.get('front_end_load_percentage', '0'))
            annual_management_fee_percentage = safe_float(entry.get('annual_management_fee_percentage', '0'))

            financials_result = CalculateTransactionFinancials(

                product_category=product_category,

            cooling_off_result = ManageCoolingOffPeriod(

            straight_through_processing_flag = stp_result['straight_through_processing_flag']
            
            # =================================================================
            # Phase 5.5: Post-Execution Compliance Review and Documentation
            # =================================================================
            
            # Step 5.5.1 & 5.5.2: Execute Post-Trade Compliance Review
            audio_video_recording_id = entry.get('audio_video_recording_id', '')
            disclosure_documents_presented = safe_json(entry.get('disclosure_documents_presented', '[]'), [])
            
            compliance_review = ExecutePostTradeComplianceReview(

                transaction_threshold_exceeded=transaction_threshold_exceeded,

            compliance_defect_flag = compliance_review['compliance_defect_flag']
            compliance_review_notes = compliance_review['compliance_review_notes']
            
            # Step 5.5.5: Calculate Processing Metrics
            if execution_timestamp:
                metrics_result = CalculateProcessingMetrics(
                    transaction_timestamp=transaction_timestamp,
                    execution_timestamp=execution_timestamp
                )
                
                total_processing_time_minutes = metrics_result['total_processing_time_minutes']
                processing_efficiency_rating = metrics_result['processing_efficiency_rating']
            else:
                total_processing_time_minutes = 0
                processing_efficiency_rating = 'Unknown'
            
            # Determine verification result
            if transaction_status == "Executed":
                verification_result = 'Transaction Successfully Executed'
            elif not pre_trade_compliance_passed:
                verification_result = 'Transaction Failed Pre-Trade Compliance'
            else:
                verification_result = 'Transaction Pending'
            
            # Update result with all output fields
            result.update({
                'calculated_risk_profile': calculated_risk_profile,
                'risk_profile_expiry_date': risk_profile_expiry_date,
                'investable_status': investable_status,
                'suitability_check_passed': str(suitability_check_passed),
                'suitability_mismatch_flag': str(suitability_mismatch_flag),
                'pre_trade_compliance_passed': str(pre_trade_compliance_passed),
                'funds_availability_verified': str(funds_availability_verified),
                'estimated_units': str(estimated_units) if estimated_units is not None else '',
                'front_end_load_amount': str(front_end_load_amount) if front_end_load_amount is not None else '',
                'net_investment_amount': str(net_investment_amount) if net_investment_amount is not None else '',
                'annual_management_fee_amount': str(annual_management_fee_amount) if annual_management_fee_amount is not None else '',
                'trade_reference_number': trade_reference_number if trade_reference_number else '',
                'contract_note_number': contract_note_number if contract_note_number else '',
                'transaction_status': transaction_status,
                'execution_timestamp': execution_timestamp,
                'settlement_date': settlement_date,
                'straight_through_processing_flag': str(straight_through_processing_flag),
                'compliance_defect_flag': str(compliance_defect_flag),
                'compliance_review_notes': compliance_review_notes,
                'supervisor_approval_timestamp': entry.get('supervisor_approval_timestamp', ''),
                'total_processing_time_minutes': str(int(total_processing_time_minutes)) if total_processing_time_minutes else ''
            })
            
            result['verification_result'] = verification_result
            result['processing_efficiency_rating'] = processing_efficiency_rating
        
    except Exception as e:
        result['verification_result'] = f'Error: {str(e)}'
        result['processing_efficiency_rating'] = 'Error'
    
    return result
