def convert_value(value: str, target_type: str):
    """Convert string value to appropriate type"""
    if value is None or value == '':
        return None
    
    if target_type == 'bool':
        return value.lower() == 'true'
    elif target_type == 'int':
        return int(float(value))
    elif target_type == 'float':
        return float(value)
    else:
        return value

def ExecutePrequalificationAnalysis(application_id, representative_credit_score, preliminary_gross_monthly_income, total_monthly_debt_obligations, requested_loan_amount, loan_term_months, estimated_property_value, down_payment_amount, property_type, property_occupancy, loan_purpose):
    """API 6: Execute Prequalification Analysis"""
    if representative_credit_score < 620:
        return {"prequalification_decision": "Not Qualified", "estimated_interest_rate_prequalification": 0, "max_loan_amount_prequalified": 0}
    
    # Estimate interest rate
    base_rate = 6.500
    if representative_credit_score >= 740:
        rate = base_rate
    elif representative_credit_score >= 700:
        rate = base_rate + 0.250
    else:
        rate = base_rate + 0.625
    
    # Calculate preliminary DTI
    monthly_rate = rate / 100.0 / 12.0
    n = loan_term_months
    if monthly_rate > 0:
        monthly_pi = requested_loan_amount * (monthly_rate * (1 + monthly_rate)**n) / ((1 + monthly_rate)**n - 1)
    else:
        monthly_pi = requested_loan_amount / n
    
    monthly_housing = monthly_pi + (estimated_property_value * 0.012 / 12) + (estimated_property_value * 0.005 / 12)
    dti = (monthly_housing + total_monthly_debt_obligations) / preliminary_gross_monthly_income * 100
    
    if dti <= 43:
        decision = "Qualified"
    elif dti <= 50:
        decision = "Conditionally Qualified"
    else:
        decision = "Not Qualified"
    
    max_loan = (0.43 * preliminary_gross_monthly_income - total_monthly_debt_obligations) * 200  # Simplified
    
    return {
        "prequalification_decision": decision,
        "estimated_interest_rate_prequalification": round(rate, 3),
        "max_loan_amount_prequalified": round(max_loan, 2),
        "preliminary_ltv_ratio": round((requested_loan_amount / estimated_property_value) * 100, 2)
    }

    """API 7: Verify Employment and Income"""

def VerifyAssetsAndReserves(application_id, prequalification_decision, liquid_assets, bank_statements_uploaded, down_payment_amount, requested_loan_amount, applicant_dob, verified_gross_monthly_income, property_type, property_occupancy, retirement_accounts=None):
    """API 8: Verify Assets and Reserves"""
    if prequalification_decision == "Not Qualified":
        return {"error": "Cannot verify for Not Qualified"}
    
    birth_date = datetime.strptime(applicant_dob, "%Y-%m-%d")
    age = datetime.utcnow().year - birth_date.year
    discount = 0.75 if age < 59.5 else 0.90
    
    total_assets = liquid_assets + (retirement_accounts * discount if retirement_accounts else 0)
    remaining = total_assets - down_payment_amount - (requested_loan_amount * 0.03)
    monthly_housing = requested_loan_amount * 0.01
    reserves = remaining / monthly_housing if monthly_housing > 0 else 0
    
    return {"reserve_months_calculated": round(reserves, 2), "verified_liquid_assets_final": round(total_assets, 2)}

def CalculateDebtRatios(application_id, verified_gross_monthly_income, total_monthly_debt_obligations, requested_loan_amount, loan_term_months, estimated_interest_rate_prequalification, property_address, estimated_property_value, representative_credit_score, preliminary_ltv_ratio):
    """API 10: Calculate Debt Ratios"""
    monthly_rate = estimated_interest_rate_prequalification / 100.0 / 12.0
    n = loan_term_months
    
    if monthly_rate > 0:
        monthly_pi = requested_loan_amount * (monthly_rate * (1 + monthly_rate)**n) / ((1 + monthly_rate)**n - 1)
    else:
        monthly_pi = requested_loan_amount / n
    
    monthly_tax = (estimated_property_value * 0.012) / 12
    monthly_insurance = (estimated_property_value * 0.005) / 12
    monthly_mi = 0
    
    if preliminary_ltv_ratio > 80:
        mi_rate = 0.007
        monthly_mi = (requested_loan_amount * mi_rate) / 12
    
    monthly_housing = monthly_pi + monthly_tax + monthly_insurance + monthly_mi
    total_obligations = monthly_housing + total_monthly_debt_obligations
    
    dti = (total_obligations / verified_gross_monthly_income) * 100
    dscr = verified_gross_monthly_income / total_obligations if total_obligations > 0 else 0
    
    return {
        "debt_to_income_ratio": round(dti, 2),
        "debt_service_coverage_ratio": round(dscr, 2),
        "monthly_housing_payment_total": round(monthly_housing, 2)
    }

def CalculateProbabilityOfDefault(application_id, representative_credit_score, debt_to_income_ratio, preliminary_ltv_ratio, reserve_months_calculated, employment_years, property_type, property_occupancy, loan_purpose, detailed_derogatory_marks_count):
    """API 11: Calculate Probability of Default"""
    z = (5.0 - 0.012 * representative_credit_score + 0.08 * debt_to_income_ratio + 
         0.05 * preliminary_ltv_ratio - 0.15 * reserve_months_calculated - 
         0.05 * employment_years + 0.2 * detailed_derogatory_marks_count)
    
    if property_type == "Multi-Family":
        z += 0.3
    if property_occupancy == "Investment Property":
        z += 0.8
    if loan_purpose == "Cash-Out Refinance":
        z += 0.4
    
    pd = (1 / (1 + math.exp(-z))) * 100
    
    if pd <= 2.0:
        grade = "A"
    elif pd <= 4.0:
        grade = "B"
    elif pd <= 7.0:
        grade = "C"
    else:
        grade = "D"
    
    return {"probability_of_default_score": round(pd, 2), "credit_score_grade": grade}

def DetermineFinalInterestRate(application_id, credit_score_grade, representative_credit_score, preliminary_ltv_ratio, property_type, property_occupancy, loan_purpose, debt_to_income_ratio, loan_term_months):

    base_rate = 6.500
    
    # Credit adjustment
    if representative_credit_score >= 780:
        credit_adj = -0.500
    elif representative_credit_score >= 740:
        credit_adj = 0.000
    elif representative_credit_score >= 700:
        credit_adj = 0.250
    elif representative_credit_score >= 680:
        credit_adj = 0.500
    else:
        credit_adj = 1.000
    
    # LTV adjustment
    if preliminary_ltv_ratio <= 80:
        ltv_adj = 0.250
    elif preliminary_ltv_ratio <= 90:
        ltv_adj = 0.625
    else:
        ltv_adj = 1.000
    
    # Property type
    prop_adj = 0.250 if property_type == "Condo" else 0.0
    
    # Occupancy
    occ_adj = 0.750 if property_occupancy == "Investment Property" else 0.0
    
    # Loan purpose
    purpose_adj = 0.500 if loan_purpose == "Cash-Out Refinance" else 0.0
    
    final_rate = base_rate + credit_adj + ltv_adj + prop_adj + occ_adj + purpose_adj
    
    return {"final_risk_adjusted_interest_rate": round(final_rate, 3)}

def EvaluateCompensatingFactors(application_id, debt_to_income_ratio, reserve_months_calculated, payment_history_quality, verified_gross_monthly_income, monthly_housing_payment_total, total_monthly_debt_obligations, preliminary_ltv_ratio, employment_years, credit_depth_rating, applicant_address):
    """API 13: Evaluate Compensating Factors"""
    points = 0
    
    if reserve_months_calculated >= 12:
        points += 3
    elif reserve_months_calculated >= 6:
        points += 2
    
    if payment_history_quality in ["Excellent", "Good"]:
        points += 2
    
    if preliminary_ltv_ratio <= 70:
        points += 2
    
    if employment_years > 5:
        points += 1
    
    return {
        "compensating_factors_present": points >= 5,
        "total_compensating_factor_points": points
    }

def OrderPropertyAppraisal(application_id, prequalification_decision, property_address, property_type, property_occupancy, estimated_property_value, requested_loan_amount, loan_purpose):
    """API 14: Order Property Appraisal"""
    if prequalification_decision == "Not Qualified":
        return {"error": "Cannot order appraisal"}
    
    return {"appraisal_order_number": f"APR-{application_id[-6:]}-001"}

def ConductPropertyValuation(application_id, appraisal_order_number, property_address, property_type, estimated_property_value):
    """API 15: Conduct Property Valuation"""
    app_data = get_application_data(application_id)
    app_number = int(application_id.split('-')[-1])
    variance = 1.0 + ((app_number % 11) - 5) / 100.0
    final_value = estimated_property_value * variance
    
    return {"final_appraised_value": round(final_value, 2)}

def ValidateAppraisalWithAVM(application_id, property_address, final_appraised_value, property_type):
    """API 16: Validate Appraisal with AVM"""
    app_number = int(application_id.split('-')[-1])
    avm_variance = 1.0 + ((app_number % 17) - 8) / 100.0
    avm_value = final_appraised_value * avm_variance
    variance_pct = ((final_appraised_value - avm_value) / avm_value) * 100
    
    return {
        "avm_estimated_value": round(avm_value, 2),
        "appraisal_avm_variance_percent": round(variance_pct, 2),
        "second_review_triggered": abs(variance_pct) > 10
    }

def CalculateLoanToValueRatio(application_id, final_appraised_value, requested_loan_amount, estimated_property_value, loan_purpose, representative_credit_score, property_type, property_occupancy):
    """API 17: Calculate Loan to Value Ratio"""
    if loan_purpose == "Purchase":
        collateral = min(final_appraised_value, estimated_property_value)
    else:
        collateral = final_appraised_value
    
    ltv = (requested_loan_amount / collateral) * 100
    
    return {"loan_to_value_ratio": round(ltv, 2)}

def RenderFinalLoanDecision(application_id, debt_to_income_ratio, debt_service_coverage_ratio, loan_to_value_ratio, probability_of_default_score, credit_score_grade, final_risk_adjusted_interest_rate, compensating_factors_present, property_condition, bankruptcy_seasoning_compliant, foreclosure_history_compliant, verified_gross_monthly_income, requested_loan_amount, loan_term_months, application_timestamp):
    """API 18: Render Final Loan Decision"""
    decline_reasons = []
    
    if property_condition in ["C4", "C5"]:
        decline_reasons.append("Property condition unsuitable")
    
    if credit_score_grade == "D":
        decline_reasons.append("Credit grade D")
    
    if debt_to_income_ratio > 50:
        decline_reasons.append("DTI exceeds 50%")
    elif debt_to_income_ratio > 43 and not compensating_factors_present:
        decline_reasons.append("DTI exceeds 43% without compensating factors")
    
    if debt_service_coverage_ratio < 1.0:
        decline_reasons.append("DSCR below 1.0")
    
    if decline_reasons:
        decision = "Declined"
        conditions = 0
        approved_amt = None
        approved_rate = None
        approved_term = None
    else:
        conditions = 0
        if debt_to_income_ratio > 43:
            conditions += 1
        if credit_score_grade == "C":
            conditions += 1
        if loan_to_value_ratio > 80:
            conditions += 1
        
        decision = "Approved with Conditions" if conditions > 0 else "Approved"
        approved_amt = requested_loan_amount
        approved_rate = final_risk_adjusted_interest_rate
        approved_term = loan_term_months
    
    app_time = datetime.strptime(application_timestamp, "%Y-%m-%dT%H:%M:%SZ")
    decision_time = datetime.utcnow()
    turnaround = (decision_time - app_time).days
    
    return {
        "final_decision": decision,
        "final_decision_timestamp": decision_time.strftime("%Y-%m-%dT%H:%M:%SZ"),
        "approved_loan_amount": approved_amt,
        "approved_interest_rate": approved_rate,
        "approved_loan_term_months": approved_term,
        "conditions_count": conditions,
        "decline_reason": "; ".join(decline_reasons) if decline_reasons else "",
        "turnaround_time_days": turnaround
    }

def ExecuteFairLendingMonitoring(application_id, final_decision, representative_credit_score, debt_to_income_ratio, loan_to_value_ratio, property_address, approved_interest_rate=None, conditions_count=None):
    """API 19: Execute Fair Lending Monitoring"""
    audit_flag = False
    if approved_interest_rate and abs(approved_interest_rate - 6.5) > 1.0:
        audit_flag = True
    
    return {"fair_lending_audit_flag": audit_flag}

def TrackLoanPerformanceMetrics(application_id, application_timestamp, final_decision, final_decision_timestamp, conditions_count, approved_loan_amount=None):
    """API 20: Track Loan Performance Metrics"""
    app_time = datetime.strptime(application_timestamp, "%Y-%m-%dT%H:%M:%SZ")
    decision_time = datetime.strptime(final_decision_timestamp, "%Y-%m-%dT%H:%M:%SZ")
    turnaround = (decision_time - app_time).days
    
    first_pass = conditions_count == 0
    funded = final_decision in ["Approved", "Approved with Conditions"]
    funding_date = (decision_time + timedelta(days=30)).strftime("%Y-%m-%d") if funded else None
    
    return {
        "turnaround_time_days": turnaround,
        "first_pass_yield": first_pass,
        "funded_status": funded,
        "funding_date": funding_date
    }

    except Exception as e:
