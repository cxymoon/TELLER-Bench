    def _parse_field(self, field_str: str) -> Tuple[str, Any]:
        """解析字段字符串为键值对"""
        parts = field_str.split(':', 1)
        if len(parts) != 2:
            return parts[0], ''
        
        key, value = parts
        
        # 处理JSON字符串
        if value.startswith('{'):
            try:
                return key, json.loads(value)
            except:
                return key, value
        
        # 处理布尔值
        if value == 'True':
            return key, True
        elif value == 'False':
            return key, False
        
        # 处理空值
        if value == '':
            return key, None
        
        # 处理数字
        try:
            if '.' in value:
                return key, float(value)
            else:
                return key, int(value)
        except:
            return key, value

    # 计算财务比率
    asset_liability_ratio = financial_statements['total_liabilities'] / financial_statements['total_assets']
    roe = financial_statements['net_profit'] / (financial_statements['total_assets'] - financial_statements['total_liabilities'])
    
    # 基础分数
    base_score = 70.0
    
    # 财务指标调整
    if asset_liability_ratio < 0.5:
        base_score += 10.0
    elif asset_liability_ratio > 0.75:
        base_score -= 10.0
    
    if roe > 0.15:
        base_score += 8.0
    elif roe < 0.05:
        base_score -= 8.0
    
    # 征信调整
    if credit_report['overdue_count'] == 0:
        base_score += 5.0
    else:
        base_score -= credit_report['overdue_count'] * 3.0
    
    # 客户关系调整
    if customer_profile['existence_months'] > 36:
        base_score += 5.0
    
    credit_score = round(max(0.0, min(100.0, base_score)), 1)
    
    return {"credit_score": credit_score}


def API_03_AssessDebtRepaymentCapacity(financial_statements: Dict, requested_amount: float,
                                        requested_term_months: int, selected_interest_rate: float,
                                        industry_category: str) -> Dict:
    """评估偿债能力"""
    # 估算EBIT (息税前利润)
    ebit = financial_statements['net_profit'] * 1.3
    
    # 计算年化还款额 (等额本息)
    monthly_rate = selected_interest_rate / 100 / 12
    monthly_payment = requested_amount * monthly_rate * (1 + monthly_rate) ** requested_term_months / \
                     ((1 + monthly_rate) ** requested_term_months - 1)
    annual_payment = monthly_payment * 12
    
    # 计算DSCR
    dscr = ebit / annual_payment if annual_payment > 0 else 0
    
    # 判定等级
    if dscr >= 1.5:
        capacity_level = "强"
    elif dscr >= 1.2:
        capacity_level = "中"
    else:
        capacity_level = "弱"
    
    return {
        "debt_repayment_capacity": round(dscr, 2),
        "capacity_level": capacity_level,
        "ebit": round(ebit, 2),
        "annual_payment": round(annual_payment, 2)
    }

def API_07_CalculateInterestRate(credit_score: float, requested_term_months: int,
                                   industry_risk_level: str, collateral_info: Dict) -> Dict:
    """计算利率"""
    # 基础LPR
    lpr = 3.65
    base_spread = 0.80
    
    # 信用调整
    if credit_score >= 85:
        credit_adjustment = -0.20
    elif credit_score >= 75:
        credit_adjustment = 0.0
    elif credit_score >= 65:
        credit_adjustment = 0.30
    else:
        credit_adjustment = 0.50
    
    # 期限调整
    term_adjustment = (requested_term_months // 12) * 0.10
    
    # 行业调整
    industry_adjustment_map = {"low": -0.10, "medium": 0.0, "high": 0.50}
    industry_adjustment = industry_adjustment_map.get(industry_risk_level, 0.0)
    
    # 担保调整
    collateral_adjustment_map = {
        "property": -0.30,
        "equipment": -0.20,
        "guarantee": -0.15,
        "none": 0.40
    }
    collateral_adjustment = collateral_adjustment_map.get(collateral_info['collateral_type'], 0.0)
    
    # 计算最终利率
    selected_interest_rate = lpr + base_spread + credit_adjustment + term_adjustment + \
                             industry_adjustment + collateral_adjustment
    
    return {
        "selected_interest_rate": round(selected_interest_rate, 2),
        "rate_components": {
            "lpr": lpr,
            "base_spread": base_spread,
            "credit_adjustment": credit_adjustment,
            "term_adjustment": term_adjustment,
            "industry_adjustment": industry_adjustment,
            "collateral_adjustment": collateral_adjustment
        }
    }


def API_08_GenerateAmortizationSchedule(requested_amount: float, requested_term_months: int,
                                         selected_interest_rate: float, 
                                         fund_disbursement_date: Optional[str] = None) -> Dict:
    """生成还款计划"""
    # 确定起始日期
    if fund_disbursement_date:
        start_date = datetime.fromisoformat(fund_disbursement_date.replace('Z', '+00:00'))
    else:
        start_date = datetime.now()
    
    # 计算月供
    monthly_rate = selected_interest_rate / 100 / 12
    monthly_payment = requested_amount * monthly_rate * (1 + monthly_rate) ** requested_term_months / \
                     ((1 + monthly_rate) ** requested_term_months - 1)
    
    schedule = []
    remaining_principal = requested_amount
    
    for period in range(1, requested_term_months + 1):
        interest = remaining_principal * monthly_rate
        principal = monthly_payment - interest
        remaining_principal -= principal
        
        payment_date = start_date + timedelta(days=30 * period)
        
        schedule.append({
            "period": period,
            "payment_date": payment_date.strftime('%Y-%m-%d'),
            "principal": round(principal, 2),
            "interest": round(interest, 2),
            "total_payment": round(monthly_payment, 2),
            "remaining_principal": round(max(0, remaining_principal), 2)
        })
    
    return {"amortization_schedule": schedule}

def API_11_ExecutePDModel(customer_id: str, credit_score: float, financial_statements: Dict,
                           industry_risk_level: str, region: str, collateral_info: Dict,
                           contract_terms_confirmed: bool) -> Dict:
    """执行违约概率模型"""
    if not contract_terms_confirmed:
        return {"error": "合同条款未确认,无法执行深度风险评估"}
    
    # 基础PD
    base_pd = 0.02
    
    # 信用评分调整
    if credit_score >= 85:
        pd_adjustment = -0.010
    elif credit_score >= 75:
        pd_adjustment = -0.005
    elif credit_score < 65:
        pd_adjustment = 0.015
    else:
        pd_adjustment = 0.0
    
    # 行业风险调整
    industry_pd_map = {"low": -0.003, "medium": 0.005, "high": 0.020}
    industry_pd = industry_pd_map.get(industry_risk_level, 0.0)
    
    # 财务状况调整
    leverage_ratio = financial_statements['total_liabilities'] / financial_statements['total_assets']
    if leverage_ratio > 0.7:
        leverage_pd = 0.010
    else:
        leverage_pd = -0.005
    
    # 担保调整
    collateral_pd_map = {"property": -0.008, "equipment": -0.005, "guarantee": -0.003, "none": 0.010}
    collateral_pd = collateral_pd_map.get(collateral_info['collateral_type'], 0.0)
    
    # 计算最终PD
    probability_of_default = base_pd + pd_adjustment + industry_pd + leverage_pd + collateral_pd
    probability_of_default = max(0.001, min(0.300, probability_of_default))
    
    return {
        "probability_of_default": round(probability_of_default, 3),
        "pd_components": {
            "base_pd": base_pd,
            "credit_adjustment": pd_adjustment,
            "industry_adjustment": industry_pd,
            "leverage_adjustment": leverage_pd,
            "collateral_adjustment": collateral_pd
        }
    }


def API_12_ExecuteLGDModel(collateral_info: Dict, requested_amount: float, 
                            industry_category: str) -> Dict:
    """执行违约损失率模型"""
    # 基础LGD
    base_lgd_map = {
        "property": 0.30,
        "equipment": 0.45,
        "guarantee": 0.55,
        "none": 0.70
    }
    base_lgd = base_lgd_map.get(collateral_info['collateral_type'], 0.70)
    
    # LTV调整
    if collateral_info['collateral_value'] > 0:
        ltv = requested_amount / collateral_info['collateral_value']

            ltv_adjustment = 0.10

    # 计算最终LGD
    loss_given_default = base_lgd + ltv_adjustment
    loss_given_default = max(0.10, min(0.90, loss_given_default))
    
    return {

def API_14_RunMonteCarloSimulation(probability_of_default: float, loss_given_default: float,
                                     exposure_at_default: float, requested_amount: float) -> Dict:
    """运行蒙特卡洛模拟"""
    import random
    
    # 模拟10000次
    simulations = 10000
    losses = []
    
    for _ in range(simulations):
        # 模拟违约事件
        default_occurred = random.random() < probability_of_default
        if default_occurred:
            # 模拟损失率波动
            simulated_lgd = max(0, min(1, random.gauss(loss_given_default, 0.1)))
            loss = exposure_at_default * simulated_lgd
        else:
            loss = 0
        losses.append(loss)
    
    # 计算统计量
    losses.sort()
    expected_loss = sum(losses) / len(losses)
    unexpected_loss_95 = losses[int(len(losses) * 0.95)]
    
    # 计算风险评分
    total_risk = expected_loss + unexpected_loss_95
    risk_score = max(0, min(100, 100 - (total_risk / requested_amount) * 100))
    
    return {
        "expected_credit_loss": round(expected_loss, 2),
        "unexpected_loss_95": round(unexpected_loss_95, 2),
        "risk_score": round(risk_score, 1),
        "simulation_count": simulations
    }

def API_20_AssignRiskRating(expected_credit_loss: float, unexpected_loss_95: float,
                              requested_amount: float, debt_repayment_capacity: float,
                              collateral_info: Dict, related_party_risk_flag: bool,
                              concentration_risk_flag: bool, collateral_duplication_flag: bool,
                              aml_suspicious_flag: bool) -> Dict:
    """分配风险评级"""
    # 计算风险评分
    total_risk = expected_credit_loss + unexpected_loss_95
    risk_score = max(0, min(100, 100 - (total_risk / requested_amount) * 100))
    
    # 计算LTV
    ltv = requested_amount / collateral_info['collateral_value'] if collateral_info['collateral_value'] > 0 else 1.0
    
    # 评级规则

def API_52_CalculateInvestmentReportDate(business_type: str, investment_return_cycle: str,
                                           fund_disbursement_date: str) -> Dict:
    """计算投资报告日期"""
    if business_type != "investment":
        return {"error": "仅投资类业务需要计算投资报告日期"}
    
    disbursement_date = datetime.fromisoformat(fund_disbursement_date.replace('Z', '+00:00'))
    
    # 根据分红周期确定基准日
    if "季度" in investment_return_cycle:
        # 下一个季度末
        current_quarter = (disbursement_date.month - 1) // 3
        next_quarter_end_month = (current_quarter + 1) * 3 + 3
        if next_quarter_end_month > 12:
            next_quarter_end_month = 3
            year = disbursement_date.year + 1
        else:
            year = disbursement_date.year
        
        # 季度末最后一天
        if next_quarter_end_month in [3, 6, 9, 12]:
            base_date = datetime(year, next_quarter_end_month, 30 if next_quarter_end_month in [6, 9] else 31)
    
    elif "月" in investment_return_cycle:
        # 下个月末
        if disbursement_date.month == 12:
            base_date = datetime(disbursement_date.year + 1, 1, 31)
        else:
            next_month = disbursement_date.month + 1
            # 简化:假设每月30日
            base_date = datetime(disbursement_date.year, next_month, 30)
    
    elif "年" in investment_return_cycle:
        # 下一年末
        base_date = datetime(disbursement_date.year + 1, 12, 31)
    
    else:
        base_date = disbursement_date + timedelta(days=365)
    
    # 基准日后第10个工作日
    next_investment_report_date = (base_date + timedelta(days=10)).strftime('%Y-%m-%d')
    
    return {
        "next_investment_report_date": next_investment_report_date,
        "return_cycle": investment_return_cycle,
        "base_date": base_date.strftime('%Y-%m-%d')
    }

    print(f"API_05 - 资格预审: {'通过' if result_05['qualification_precheck_result'] else '不通过'}")
    print(f"         原因: {result_05['precheck_reasons']}")
    data_store.update_application(application_id, {"qualification_precheck_result": result_05['qualification_precheck_result']})

        print("\n资格预审未通过,流程终止")
        return

        "requested_amount": app_data['requested_amount'],
        "requested_term_months": app_data['requested_term_months'],
        "selected_interest_rate": app_data['selected_interest_rate'],
        "fund_disbursement_date": None
    })
    print(f"API_08 - 还款计划: 生成{len(result_08['amortization_schedule'])}期还款计划")

    # 阶段3: 风险评估
    print("\n【阶段3: 风险管理深度评估】")
    
    # API_11: PD模型
    result_11 = process_tool_call("API_11_ExecutePDModel", {

        "contract_terms_confirmed": app_data['contract_terms_confirmed']
    })
    print(f"API_11 - 违约概率(PD): {result_11['probability_of_default']}")

    result_12 = process_tool_call("API_12_ExecuteLGDModel", {
        "collateral_info": app_data['collateral_info'],
        "requested_amount": app_data['requested_amount'],
        "industry_category": app_data['industry_category']
    })
    print(f"API_12 - 违约损失率(LGD): {result_12['loss_given_default']}")
    
    # API_13: EAD模型
    result_13 = process_tool_call("API_13_ExecuteEADModel", {
        "business_type": app_data['business_type'],
        "requested_amount": app_data['requested_amount'],
        "customer_id": app_data['customer_id'],
        "loan_history": result_01['loan_history']
    })
    print(f"API_13 - 风险敞口(EAD): {result_13['exposure_at_default']}")
    
    # API_14: 蒙特卡洛模拟
    result_14 = process_tool_call("API_14_RunMonteCarloSimulation", {
        "probability_of_default": result_11['probability_of_default'],
        "loss_given_default": result_12['loss_given_default'],

        "requested_amount": app_data['requested_amount']
    })
    print(f"API_14 - ECL模拟: 预期损失={result_14['expected_credit_loss']}, 风险评分={result_14['risk_score']}")
    
    # API_20: 风险评级
    result_20 = process_tool_call("API_20_AssignRiskRating", {
        "expected_credit_loss": result_14['expected_credit_loss'],
        "unexpected_loss_95": result_14['unexpected_loss_95'],
        "requested_amount": app_data['requested_amount'],
        "debt_repayment_capacity": result_03['debt_repayment_capacity'],

        "related_party_risk_flag": False,
        "concentration_risk_flag": False,
        "collateral_duplication_flag": False,
        "aml_suspicious_flag": False
    })
    print(f"API_20 - 风险评级: {result_20['risk_rating']}, 评分={result_20['risk_score']}")
    
    # 阶段5: 审批
    print("\n【阶段5: 审批决策】")
    
    # API_33: 执行审批
    result_33 = process_tool_call("API_33_ExecuteApprovalDecision", {
        "approval_level": "risk_assessor",
        "approver_ids": ["ASSESSOR_001"],
        "application_id": application_id,

    })
    print(f"API_33 - 审批决策: {result_33['final_approval_status']}")
    
    # API_34: 记录审批结果
    result_34 = process_tool_call("API_34_RecordApprovalResult", {

        "approval_timestamp": result_33['approval_timestamp'],
        "application_id": application_id,
        "customer_id": app_data['customer_id'],
        "requested_amount": app_data['requested_amount']
    })
    print(f"API_34 - 审批记录: {result_34['approval_record_id']}")
    
    print("\n✅ 贷款业务测试完成")
    print(f"最终数据状态: {json.dumps(data_store.get_by_application_id(application_id), ensure_ascii=False, indent=2)}")
