APPLICATION_DATA = []

def parse_data_entry(entry: List[str]) -> Dict[str, Any]:
    """Parse a data entry list into a dictionary"""
    data_dict = {}
    for item in entry:
        if ':' in item:
            key, value = item.split(':', 1)
            if value == '':
                data_dict[key] = None
            elif value in ['True', 'False']:
                data_dict[key] = value == 'True'
            elif value.startswith('['):
                try:
                    data_dict[key] = json.loads(value.replace("'", '"'))
                except:
                    data_dict[key] = []
            else:
                try:
                    if '.' not in value:
                        data_dict[key] = int(value)
                    else:
                        data_dict[key] = float(value)
                except:
                    data_dict[key] = value
    return data_dict

def initialize_data(raw_data: List[List[str]]):
    """Initialize the APPLICATION_DATA from raw data"""
    global APPLICATION_DATA
    APPLICATION_DATA = [parse_data_entry(entry) for entry in raw_data]

def get_application_by_id(application_id: str) -> Optional[Dict[str, Any]]:
    """Retrieve application data by application_id"""
    for app in APPLICATION_DATA:
        if app.get('application_id') == application_id:
            return app
    return None

def update_application_data(application_id: str, updates: Dict[str, Any]):
    """Update application data with new fields"""
    app = get_application_by_id(application_id)
    if app:
        app.update(updates)

def calculate_timestamp_diff_seconds(start_ts: str, end_ts: str) -> int:
    """Calculate difference between two ISO 8601 timestamps in seconds"""
    start = datetime.fromisoformat(start_ts.replace('Z', '+00:00'))
    end = datetime.fromisoformat(end_ts.replace('Z', '+00:00'))
    return int((end - start).total_seconds())

def add_business_days(start_date: datetime, days: int) -> str:
    """Add business days to a date (excluding weekends)"""
    current = start_date
    added = 0
    while added < days:
        current += timedelta(days=1)
        if current.weekday() < 5:
            added += 1
    return current.strftime('%Y-%m-%d')

# Process each application
for app in APPLICATION_DATA:
    try:
        application_id = app['application_id']
        
        # Step 1: Validate Application Data (Section 5.1)
        validation_result = ValidateApplicationData(
            application_id=app['application_id'],
            application_timestamp=app['application_timestamp'],
            application_channel=app['application_channel'],
            business_type=app['business_type'],
            product_type=app['product_type'],
            applicant_id=app['applicant_id'],
            applicant_age=app['applicant_age'],
            applicant_employment_status=app['applicant_employment_status'],
            applicant_annual_income=app['applicant_annual_income'],
            applicant_housing_status=app['applicant_housing_status'],
            applicant_monthly_housing_cost=app['applicant_monthly_housing_cost'],
            applicant_years_at_address=app['applicant_years_at_address'],
            applicant_years_employed=app['applicant_years_employed'],
            business_entity_name=app.get('business_entity_name'),
            business_tin_ein=app.get('business_tin_ein'),
            business_incorporation_date=app.get('business_incorporation_date'),
            business_annual_revenue=app.get('business_annual_revenue'),
            business_industry_code=app.get('business_industry_code'),
            guarantor_personal_income=app.get('guarantor_personal_income'),
            security_deposit_amount=app.get('security_deposit_amount'),
            deposit_verification_status=app.get('deposit_verification_status'),
            deposit_account_id=app.get('deposit_account_id')
        )
        
        if not validation_result['validation_status']:
            continue
        
        # Step 2: Verify Identity (Section 5.2)
        identity_result = VerifyIdentity(
            application_id=application_id,
            applicant_id=app['applicant_id'],
            applicant_age=app['applicant_age'],
            device_fingerprint_id=app['device_fingerprint_id'],
            ip_address_location=app['ip_address_location'],
            fraud_alert_flags=app.get('fraud_alert_flags', [])
        )
        
        # Step 3: Screen Sanctions Watchlist (Section 5.2)
        sanctions_result = ScreenSanctionsWatchlist(
            application_id=application_id,
            applicant_id=app['applicant_id'],
            identity_verification_method=identity_result['identity_verification_method'],
            business_entity_name=app.get('business_entity_name')
        )
        
        # Step 4: Detect Fraud Indicators (Section 5.2)
        fraud_result = DetectFraudIndicators(
            application_id=application_id,
            watchlist_check_result=sanctions_result['watchlist_check_result'],
            device_fingerprint_id=app['device_fingerprint_id'],
            ip_address_location=app['ip_address_location'],
            fraud_alert_flags=app.get('fraud_alert_flags', [])
        )
        
        # Step 5: Retrieve Credit Bureau Data (Section 5.3)
        credit_result = RetrieveCreditBureauData(
            application_id=application_id,
            applicant_id=app['applicant_id'],
            business_type=app['business_type'],
            fraud_risk_level=fraud_result['fraud_risk_level'],
            business_tin_ein=app.get('business_tin_ein')
        )
        
        # Step 6: Calculate Debt-to-Income Ratio (Section 5.4)
        dti_result = CalculateDebtToIncomeRatio(
            application_id=application_id,
            business_type=app['business_type'],
            applicant_annual_income=app['applicant_annual_income'],
            applicant_monthly_housing_cost=app['applicant_monthly_housing_cost'],
            total_outstanding_debt=credit_result['total_outstanding_debt'],
            business_annual_revenue=app.get('business_annual_revenue'),
            guarantor_personal_income=app.get('guarantor_personal_income')
        )
        
        # Step 7: Evaluate Credit Scorecard (Section 5.5)
        scorecard_result = EvaluateCreditScorecard(
            application_id=application_id,
            business_type=app['business_type'],
            credit_score=credit_result['credit_score'],
            credit_utilization_ratio=credit_result['credit_utilization_ratio'],
            debt_to_income_ratio=dti_result['debt_to_income_ratio'],
            number_of_open_accounts=credit_result['number_of_open_accounts'],
            number_of_delinquent_accounts=credit_result['number_of_delinquent_accounts'],
            bankruptcy_flag=credit_result['bankruptcy_flag'],
            applicant_years_employed=app['applicant_years_employed'],
            applicant_years_at_address=app['applicant_years_at_address'],
            months_since_last_delinquency=credit_result['months_since_last_delinquency']
        )
        
        # Step 8: Determine Credit Decision (Section 5.6)
        decision_result = DetermineCreditDecision(
            application_id=application_id,
            business_type=app['business_type'],
            product_type=app['product_type'],
            internal_risk_score=scorecard_result['internal_risk_score'],
            scorecard_reason_codes=scorecard_result['scorecard_reason_codes'],
            watchlist_check_result=sanctions_result['watchlist_check_result'],
            identity_verification_method=identity_result['identity_verification_method'],
            bankruptcy_flag=credit_result['bankruptcy_flag'],
            fraud_alert_flags=fraud_result['fraud_alert_flags'],
            applicant_annual_income=app['applicant_annual_income'],
            business_industry_code=app.get('business_industry_code')
        )
        
        # Step 9: Assign Credit Terms (Section 5.7) - Only for approved applications
        if decision_result['final_decision'] == 'approved':
            terms_result = AssignCreditTerms(
                application_id=application_id,
                final_decision=decision_result['final_decision'],
                business_type=app['business_type'],
                product_type=app['product_type'],
                internal_risk_score=scorecard_result['internal_risk_score'],
                applicant_annual_income=app['applicant_annual_income'],
                business_annual_revenue=app.get('business_annual_revenue'),
                security_deposit_amount=app.get('security_deposit_amount'),
                deposit_verification_status=app.get('deposit_verification_status')
            )
        
        # Step 10: Generate Adverse Action Notice (Section 5.9) - Only for declined
        if decision_result['final_decision'] == 'declined':
            adverse_result = GenerateAdverseActionNotice(
                application_id=application_id,
                final_decision=decision_result['final_decision'],
                decline_reason_code=decision_result['decline_reason_code'],
                applicant_id=app['applicant_id'],
                credit_bureau_source=credit_result['credit_bureau_source'],
                application_timestamp=app['application_timestamp']
            )
        
        # Step 11: Finalize Application Outputs (Section 5.8)
        final_result = FinalizeApplicationOutputs(
            application_id=application_id,
            processing_start_timestamp=app['processing_start_timestamp'],
            final_decision=decision_result['final_decision'],
            adverse_action_letter_generated=app.get('adverse_action_letter_generated', False),
            manual_review_required=decision_result.get('manual_review_required', app.get('manual_review_required', False)),
            fraud_alert_flags=fraud_result['fraud_alert_flags'],
            uploaded_documents=app.get('uploaded_documents', []),
            approved_credit_limit=app.get('approved_credit_limit'),
            assigned_apr=app.get('assigned_apr'),
            underwriter_id=app.get('underwriter_id'),
            deposit_verification_status=app.get('deposit_verification_status')
        )
        
    except Exception as e:
        print(f"Error processing {application_id}: {str(e)}")
        continue
